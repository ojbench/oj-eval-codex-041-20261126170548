cmake_minimum_required(VERSION 3.10)
project(sjtu_map LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(code tests/map_test.cpp)

target_compile_options(code PRIVATE -O2 -Wall -Wextra -UNDEBUG)
enable_testing()
add_test(NAME map_differential COMMAND code)
